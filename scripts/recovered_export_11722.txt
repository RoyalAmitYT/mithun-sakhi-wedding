/**
 * Official Wedding Invitation Export & Download Utilities
 * Pure code-generated export system for Mithun & Sakhi Wedding.
 * Zero raster image dependencies. 100% Vector & High-Resolution Canvas Rendering.

import { generateQrMatrix } from "./qr-code";

export const CANONICAL_PUBLIC_INVITATION_URL =
  "https://mithun-sakhi-wedding.vercel.app/invitation";
export const CANONICAL_INVITATION_PNG_FILENAME =
  "mithun-sakhi-official-wedding-invitation.png";

export interface InvitationExportOptions {
  scale?: number;
  filename?: string;
  canonicalUrl?: string;
}

/**
 * High-Density Canvas Renderer for the Official Mithun & Sakhi Wedding Invitation.
 * Renders at ultra-high print resolution (default 1920 x 2520 at 3x scale) with
 * architectural burgundy corner brackets, multi-layer gold borders, top ogee arch crest,
 * luxury display typography, burgundy date plaque, 2x2 ceremony cards, and genuine scannable QR code.
 */
export function renderInvitationToCanvas(
  canvas: HTMLCanvasElement,
  options: InvitationExportOptions = {}
): void {
  const scale = options.scale ?? 3;
  const baseWidth = 640;
  const baseHeight = 840;

  canvas.width = baseWidth * scale;
  canvas.height = baseHeight * scale;

  const ctx = canvas.getContext("2d");
  if (!ctx) return;

  ctx.scale(scale, scale);

  // 1. Rich Warm Ivory Parchment Background with Rounded Card Base
  const bgGrad = ctx.createLinearGradient(0, 0, 0, baseHeight);
  bgGrad.addColorStop(0, "#FFFDF9");
  bgGrad.addColorStop(0.5, "#FAF5EC");
  bgGrad.addColorStop(1, "#F5EFEB");

  ctx.save();
  ctx.beginPath();
  ctx.roundRect(0, 0, baseWidth, baseHeight, 32);
  ctx.clip();

  ctx.fillStyle = bgGrad;
  ctx.fillRect(0, 0, baseWidth, baseHeight);

  // 2. Four Regal Burgundy Cusped Corner Bracket Panels
  const drawCornerPanel = (cx: number, cy: number, flipX: number, flipY: number) => {
    ctx.save();
    ctx.translate(cx, cy);
    ctx.scale(flipX, flipY);

    // Burgundy Corner Fill
    const cornerGrad = ctx.createLinearGradient(0, 0, 80, 80);
    cornerGrad.addColorStop(0, "#4A0A10");
    cornerGrad.addColorStop(0.5, "#6E141E");
    cornerGrad.addColorStop(1, "#550E17");
    ctx.fillStyle = cornerGrad;

    ctx.beginPath();
    ctx.moveTo(0, 0);
    ctx.lineTo(84, 0);
    ctx.bezierCurveTo(84, 25, 68, 44, 48, 48);
    ctx.bezierCurveTo(44, 68, 25, 84, 0, 84);
    ctx.closePath();
    ctx.fill();

    // Outer Gold Contour
    ctx.strokeStyle = "#D4AF37";
    ctx.lineWidth = 1.8;
    ctx.beginPath();
    ctx.moveTo(84, 0);
    ctx.bezierCurveTo(84, 25, 68, 44, 48, 48);
    ctx.bezierCurveTo(44, 68, 25, 84, 0, 84);
    ctx.stroke();

    // Inner Gold Hairline
    ctx.strokeStyle = "rgba(245, 230, 163, 0.75)";
    ctx.lineWidth = 0.8;
    ctx.beginPath();
    ctx.moveTo(78, 0);
    ctx.bezierCurveTo(78, 22, 63, 39, 44, 44);
    ctx.bezierCurveTo(39, 63, 22, 78, 0, 78);
    ctx.stroke();

    // Finial Cusp Bead
    ctx.fillStyle = "#D4AF37";
    ctx.beginPath();
    ctx.arc(48, 48, 2.2, 0, Math.PI * 2);
    ctx.fill();

    ctx.restore();
  };

  drawCornerPanel(0, 0, 1, 1);
  drawCornerPanel(baseWidth, 0, -1, 1);
  drawCornerPanel(0, baseHeight, 1, -1);
  drawCornerPanel(baseWidth, baseHeight, -1, -1);

  // 3. Multi-Layered Royal Gold Borders
  // Outer hairline border
  ctx.strokeStyle = "rgba(212, 175, 55, 0.35)";
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.roundRect(10, 10, baseWidth - 20, baseHeight - 20, 24);
  ctx.stroke();

  // Primary double gold border
  ctx.strokeStyle = "rgba(212, 175, 55, 0.75)";
  ctx.lineWidth = 1.5;
  ctx.beginPath();
  ctx.roundRect(16, 16, baseWidth - 32, baseHeight - 32, 18);
  ctx.stroke();

  // Inner dashed gold hairline
  ctx.strokeStyle = "rgba(216, 184, 120, 0.35)";
  ctx.lineWidth = 0.8;
  ctx.setLineDash([3, 3]);
  ctx.beginPath();
  ctx.roundRect(22, 22, baseWidth - 44, baseHeight - 44, 14);
  ctx.stroke();
  ctx.setLineDash([]);

  // 4. TOP: Ogee Arch Crest with Monogram: M&S
  const crestY = 32;
  const crestW = 90;
  const crestH = 40;
  const crestX = baseWidth / 2 - crestW / 2;

  // Flanking horizontal gold lines with beads
  ctx.strokeStyle = "rgba(212, 175, 55, 0.6)";
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(110, crestY + crestH / 2);
  ctx.lineTo(crestX - 4, crestY + crestH / 2);
  ctx.moveTo(crestX + crestW + 4, crestY + crestH / 2);
  ctx.lineTo(baseWidth - 110, crestY + crestH / 2);
  ctx.stroke();

  ctx.fillStyle = "#D4AF37";
  ctx.beginPath();
  ctx.arc(crestX - 12, crestY + crestH / 2, 2, 0, Math.PI * 2);
  ctx.arc(crestX + crestW + 12, crestY + crestH / 2, 2, 0, Math.PI * 2);
  ctx.fill();

  // Medallion Base
  ctx.fillStyle = "#FAF5EC";
  ctx.strokeStyle = "#D4AF37";
  ctx.lineWidth = 1.6;
  ctx.beginPath();
  ctx.roundRect(crestX, crestY, crestW, crestH, 14);
  ctx.fill();
  ctx.stroke();

  // Monogram M&S
  ctx.fillStyle = "#5A0F17";
  ctx.font = "bold 18px Georgia, serif";
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.fillText("M & S", baseWidth / 2, crestY + crestH / 2);

  // 5. Divine Invocation Statement
  ctx.fillStyle = "#6E141E";
  ctx.font = "bold 11px -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif";
  ctx.textAlign = "center";
  ctx.letterSpacing = "2.5px";
  ctx.fillText("WITH DIVINE BLESSINGS OF OUR ANCESTORS AND FAMILIES", baseWidth / 2, 102);

  // Geometric Divider
  const drawDivider = (y: number) => {
    ctx.strokeStyle = "rgba(212, 175, 55, 0.5)";
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(baseWidth / 2 - 90, y);
    ctx.lineTo(baseWidth / 2 - 16, y);
    ctx.moveTo(baseWidth / 2 + 16, y);
    ctx.lineTo(baseWidth / 2 + 90, y);
    ctx.stroke();

    ctx.fillStyle = "#801C2D";
    ctx.font = "bold 9px sans-serif";
    ctx.textAlign = "center";
    ctx.textBaseline = "middle";
    ctx.fillText("◆", baseWidth / 2, y);

    ctx.fillStyle = "#D4AF37";
    ctx.beginPath();
    ctx.arc(baseWidth / 2 - 8, y, 1.5, 0, Math.PI * 2);
    ctx.arc(baseWidth / 2 + 8, y, 1.5, 0, Math.PI * 2);
    ctx.fill();
  };

  drawDivider(118);

  // 6. Parents Attribution
  const parentsY = 132;
  const pColOffset = 130;

  // Groom's Parents Capsule
  ctx.fillStyle = "rgba(250, 243, 230, 0.85)";
  ctx.strokeStyle = "rgba(212, 175, 55, 0.5)";
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.roundRect(baseWidth / 2 - pColOffset - 65, parentsY, 130, 20, 10);
  ctx.fill();
  ctx.stroke();

  ctx.fillStyle = "#801C2D";
  ctx.font = "bold 9px -apple-system, sans-serif";
  ctx.textAlign = "center";
  ctx.fillText("GROOM'S PARENTS", baseWidth / 2 - pColOffset, parentsY + 13);

  ctx.fillStyle = "#292322";
  ctx.font = "bold 13.5px Georgia, serif";
  ctx.fillText("Mr. Kenaram Gorain", baseWidth / 2 - pColOffset, parentsY + 36);
  ctx.fillText("& Ms. Savita Gorian", baseWidth / 2 - pColOffset, parentsY + 52);

  ctx.fillStyle = "#6B5C57";
  ctx.font = "10.5px -apple-system, sans-serif";
  ctx.fillText("Chitmu, Kotshila, Purulia", baseWidth / 2 - pColOffset, parentsY + 68);

  // Vertical Center Divider
  ctx.strokeStyle = "rgba(212, 175, 55, 0.4)";
  ctx.beginPath();
  ctx.moveTo(baseWidth / 2, parentsY + 10);
  ctx.lineTo(baseWidth / 2, parentsY + 65);
  ctx.stroke();

  ctx.fillStyle = "#801C2D";
  ctx.fillText("◆", baseWidth / 2, parentsY + 38);

  // Bride's Parents Capsule
  ctx.fillStyle = "rgba(250, 243, 230, 0.85)";
  ctx.strokeStyle = "rgba(212, 175, 55, 0.5)";
  ctx.beginPath();
  ctx.roundRect(baseWidth / 2 + pColOffset - 65, parentsY, 130, 20, 10);
  ctx.fill();
  ctx.stroke();

  ctx.fillStyle = "#801C2D";
  ctx.font = "bold 9px -apple-system, sans-serif";
  ctx.fillText("BRIDE'S PARENTS", baseWidth / 2 + pColOffset, parentsY + 13);

  ctx.fillStyle = "#292322";
  ctx.font = "bold 13.5px Georgia, serif";
  ctx.fillText("Mr. Upen Gorain", baseWidth / 2 + pColOffset, parentsY + 36);
  ctx.fillText("& Ms. Kokila Gorain", baseWidth / 2 + pColOffset, parentsY + 52);

  ctx.fillStyle = "#6B5C57";
  ctx.font = "10.5px -apple-system, sans-serif";
  ctx.fillText("Ghaghra, Jaipur, Purulia", baseWidth / 2 + pColOffset, parentsY + 68);

  drawDivider(216);

  // 7. Invitation Statement Decree
  ctx.fillStyle = "#42332E";
  ctx.font = "italic 14.5px Georgia, serif";
  ctx.textAlign = "center";
  ctx.fillText(
    "Together with their families, Mithun & Sakhi joyfully invite you to celebrate their sacred union.",
    baseWidth / 2,
    236
  );

  // 8. CENTERPIECE: Couple Names
  const coupleY = 276;
  ctx.fillStyle = "#68131B";
  ctx.font = "italic 36px Georgia, serif";
  ctx.fillText("Mithun Gorain", baseWidth / 2, coupleY);

  // Ampersand Flourish
  ctx.fillStyle = "#AA822A";
  ctx.font = "italic 22px Georgia, serif";
  ctx.fillText("&", baseWidth / 2, coupleY + 28);

  ctx.strokeStyle = "rgba(212, 175, 55, 0.6)";
  ctx.beginPath();
  ctx.moveTo(baseWidth / 2 - 50, coupleY + 28);
  ctx.lineTo(baseWidth / 2 - 18, coupleY + 28);
  ctx.moveTo(baseWidth / 2 + 18, coupleY + 28);
  ctx.lineTo(baseWidth / 2 + 50, coupleY + 28);
  ctx.stroke();

  ctx.fillStyle = "#68131B";
  ctx.font = "italic 36px Georgia, serif";
  ctx.fillText("Sakhi Gorain", baseWidth / 2, coupleY + 62);

  drawDivider(coupleY + 84);

  // 9. Wedding Date Plaque & Location
  const plaqueY = coupleY + 98;
  const plaqueW = 280;
  const plaqueH = 34;

  const plaqueGrad = ctx.createLinearGradient(baseWidth / 2 - plaqueW / 2, 0, baseWidth / 2 + plaqueW / 2, 0);
  plaqueGrad.addColorStop(0, "#4A0A10");
  plaqueGrad.addColorStop(0.5, "#70141E");
  plaqueGrad.addColorStop(1, "#4A0A10");

  ctx.fillStyle = plaqueGrad;
  ctx.strokeStyle = "#D4AF37";
  ctx.lineWidth = 1.6;
  ctx.beginPath();
  ctx.roundRect(baseWidth / 2 - plaqueW / 2, plaqueY, plaqueW, plaqueH, 8);
  ctx.fill();
  ctx.stroke();

  ctx.fillStyle = "#FFF9EE";
  ctx.font = "bold 13px Georgia, serif";
  ctx.textAlign = "center";
  ctx.fillText("19 – 22 NOVEMBER 2026", baseWidth / 2, plaqueY + 22);

  ctx.fillStyle = "#6E141E";
  ctx.font = "bold 11px -apple-system, sans-serif";
  ctx.fillText("PURULIA, WEST BENGAL", baseWidth / 2, plaqueY + 48);

  drawDivider(plaqueY + 62);

  // 10. Four-Ceremony Grid (2x2)
  const gridY = plaqueY + 74;
  const boxW = 245;
  const boxH = 68;
  const col1 = baseWidth / 2 - boxW - 8;
  const col2 = baseWidth / 2 + 8;

  const ceremonies = [
    {
      title: "HALDI CEREMONY",
      dt: "19 November 2026 | 9:00 AM IST",
      venue: "Groom's Home, Chitmu, Kotshila, Purulia",
      bg: "#FFFDF4",
      border: "rgba(229, 201, 122, 0.85)",
      x: col1,
      y: gridY,
    },
    {
      title: "MEHNDI CEREMONY",
      dt: "20 November 2026 | 9:00 AM IST",
      venue: "Groom's Home, Chitmu, Kotshila, Purulia",
      bg: "#F6FAF6",
      border: "rgba(162, 204, 166, 0.85)",
      x: col2,
      y: gridY,
    },
    {
      title: "THE SACRED WEDDING",
      dt: "21 November 2026 | 11:00 PM IST",
      venue: "Bride's Home, Ghaghra, Jaipur, Purulia",
      bg: "#FDF6F6",
      border: "rgba(223, 159, 159, 0.85)",
      x: col1,
      y: gridY + boxH + 8,
    },
    {
      title: "RECEPTION & PREETIBHOJ",
      dt: "22 November 2026 | 5:00 PM IST",
      venue: "Groom's Home, Chitmu, Kotshila, Purulia",
      bg: "#FAF7F4",
      border: "rgba(214, 189, 168, 0.85)",
      x: col2,
      y: gridY + boxH + 8,
    },
  ];

  ceremonies.forEach((c) => {
    ctx.fillStyle = c.bg;
    ctx.strokeStyle = c.border;
    ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.roundRect(c.x, c.y, boxW, boxH, 8);
    ctx.fill();
    ctx.stroke();

    ctx.textAlign = "center";
    ctx.fillStyle = "#6E141E";
    ctx.font = "bold 11px Georgia, serif";
    ctx.fillText(c.title, c.x + boxW / 2, c.y + 19);

    ctx.fillStyle = "#292322";
    ctx.font = "bold 10px -apple-system, sans-serif";
    ctx.fillText(c.dt, c.x + boxW / 2, c.y + 36);

    ctx.fillStyle = "#6B5C57";
    ctx.font = "9.5px -apple-system, sans-serif";
    ctx.fillText(c.venue, c.x + boxW / 2, c.y + 53);
  });

  drawDivider(gridY + boxH * 2 + 18);

  // 11. BOTTOM SECTION: Real Scannable QR Code & Message
  const bottomY = gridY + boxH * 2 + 28;
  const qrUrl = options.canonicalUrl || CANONICAL_PUBLIC_INVITATION_URL;
  const qrMatrix = generateQrMatrix(qrUrl);
  const qrSize = 64;
  const qrX = baseWidth / 2 - 130;

  // QR White Base
  ctx.fillStyle = "#FFFFFF";
  ctx.strokeStyle = "rgba(212, 175, 55, 0.6)";
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.roundRect(qrX - 4, bottomY - 4, qrSize + 8, qrSize + 8, 6);
  ctx.fill();
  ctx.stroke();

  // Draw QR Modules
  const moduleCount = qrMatrix.length;
  const cellSize = qrSize / moduleCount;
  ctx.fillStyle = "#292322";

  for (let r = 0; r < moduleCount; r++) {
    for (let c = 0; c < moduleCount; c++) {
      if (qrMatrix[r][c]) {
        ctx.fillRect(qrX + c * cellSize, bottomY + r * cellSize, cellSize, cellSize);
      }
    }
  }

  // Vertical Separator
  ctx.strokeStyle = "rgba(212, 175, 55, 0.4)";
  ctx.beginPath();
  ctx.moveTo(baseWidth / 2 - 40, bottomY);
  ctx.lineTo(baseWidth / 2 - 40, bottomY + qrSize);
  ctx.stroke();

  // Message Text
  ctx.textAlign = "left";
  ctx.fillStyle = "#3D2B27";
  ctx.font = "12px Georgia, serif";
  ctx.fillText("Scan to view the digital invitation", baseWidth / 2 - 25, bottomY + 24);
  ctx.fillText("and complete wedding details.", baseWidth / 2 - 25, bottomY + 40);

  ctx.fillStyle = "#AA822A";
  ctx.font = "8px sans-serif";
  ctx.fillText("•  ◆  •", baseWidth / 2 - 25, bottomY + 54);

  ctx.restore();
}

/**
 * Triggers a direct download of the high-resolution PNG invitation.
 */
export async function downloadInvitationCardAsPng(
  options: InvitationExportOptions = {}
): Promise<void> {
  if (typeof window === "undefined" || typeof document === "undefined") return;

  const canvas = document.createElement("canvas");
  renderInvitationToCanvas(canvas, options);

  const filename = options.filename || CANONICAL_INVITATION_PNG_FILENAME;

  canvas.toBlob((blob) => {
    if (!blob) return;
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(() => URL.revokeObjectURL(url), 1000);
  }, "image/png");
}

/**
 * Backward-compatible helper for exporting the official invitation card to PNG.
 */
export async function exportInvitationCardToPng(
  options?: InvitationExportOptions
): Promise<void> {
  await downloadInvitationCardAsPng(options);
}

/**
 * Triggers browser print/PDF export preserving original aspect ratio.
 */
export function printOfficialInvitation(): void {
  if (typeof window !== "undefined") {
    window.print();
  }
}

/**
 * Handles sharing the canonical public invitation URL or copying to clipboard.
 */
export async function shareOrCopyInvitationUrl(): Promise<{ success: boolean; message: string }> {
  if (typeof window === "undefined") {
    return { success: false, message: "Browser window not available." };
  }

  const shareData = {
    title: "Mithun & Sakhi — Official Wedding Invitation",
    text: "Official royal wedding invitation decree for Mithun Gorain & Sakhi Gorain. 19–22 November 2026, Purulia, West Bengal.",
    url: CANONICAL_PUBLIC_INVITATION_URL,
  };

  if (navigator.share && navigator.canShare && navigator.canShare(shareData)) {
    try {
      await navigator.share(shareData);
      return { success: true, message: "Invitation shared successfully!" };
    } catch {
      // Fall through to clipboard copy
    }
  }

  try {
    await navigator.clipboard.writeText(CANONICAL_PUBLIC_INVITATION_URL);
    return { success: true, message: "Invitation link copied to clipboard!" };
  } catch {
    return { success: false, message: `Public URL: ${CANONICAL_PUBLIC_INVITATION_URL}` };
  }
}
